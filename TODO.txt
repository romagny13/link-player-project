en JS on pourrait avoir
class MyClasse {
  onNavigated(context) {

  }
}

function hasNavigationAware(target) {
  return typeof target?.onNavigated === "function";
}

function dispatchNavigation(viewOrController, context) {
  if (hasNavigationAware(viewOrController)) {
    viewOrController.onNavigated(context);
  }
}
